cmake_minimum_required(VERSION 3.16)
project(receiver_replay C ASM)
include(../models/assets.cmake)
add_library(receiver_core ../main/decoder.c "${assets}")
target_include_directories(receiver_core PUBLIC ../main)
target_link_libraries(receiver_core PUBLIC m)
target_compile_options(receiver_core PRIVATE -Wall -Wextra)
add_executable(replay replay.c)
target_link_libraries(replay PRIVATE receiver_core)
if(UNIX)
    add_executable(receiverctl receiverctl.c)
endif()
enable_testing()
add_executable(test_history test_history.c ../main/history.c ../main/cpr.c)
target_link_libraries(test_history PRIVATE receiver_core)
add_test(NAME history COMMAND test_history)
add_executable(test_cpr test_cpr.c ../main/cpr.c)
target_include_directories(test_cpr PRIVATE ../main)
target_link_libraries(test_cpr PRIVATE m)
add_test(NAME cpr COMMAND test_cpr)
add_executable(test_tuning test_tuning.c ../main/tuning.c)
target_include_directories(test_tuning PRIVATE ../main)
add_test(NAME tuning COMMAND test_tuning)
add_test(NAME recordings COMMAND sh "${CMAKE_CURRENT_SOURCE_DIR}/check.sh" $<TARGET_FILE:replay>)
